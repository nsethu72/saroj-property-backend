Import express from ‘express’;
Import { prisma } from ‘../index.js’;
Import { z } from ‘zod’;
Import { logAudit } from ‘../utils/helpers.js’;

Const router = express.Router();

Const createTenantSchema = z.object({
  Name: z.string(),
  residentialAddress: z.string(),
  aadhaarNumber: z.string(),
  panNumber: z.string(),
  otherIdType: z.string(),
  otherIdNumber: z.string(),
  companyName: z.string(),
  workPlace: z.string(),
  documentUrls: z.array(z.string().url()).min(2)
});

Router.get(‘/’, async (req, res) => {
  Const tenants = await prisma.tenant.findMany({
    orderBy: { createdAt: ‘desc’ }
  });
  Res.json(tenants);
});

Router.post(‘/’, async (req, res) => {
  Try {
    Const body = createTenantSchema.parse(req.body);
    Const tenant = await prisma.tenant.create({ data: body });

    Await logAudit(
      Req.user.id,
      ‘CREATE’,
      ‘Tenant’,
      Tenant.id,
      { action: ‘created’, data: body }
    );

    Res.status(201).json(tenant);
  } catch € {
    Res.status(400).json({ error: ‘Invalid input’, details: e.message });
  }
});

Router.get(‘/:id’, async (req, res) => {
  Const tenant = await prisma.tenant.findUnique({ where: { id: req.params.id } });
  If (!tenant) return res.status(404).json({ error: ‘Tenant not found’ });
  Res.json(tenant);
});

Export default router;
