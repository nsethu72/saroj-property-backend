Import express from ‘express’;
Import { prisma } from ‘../index.js’;
Import { z } from ‘zod’;
Import { logAudit } from ‘../utils/helpers.js’;

Const router = express.Router();

Const createAssetSchema = z.object({
  propertyId: z.string(),
  name: z.string(),
  category: z.string(),
  purchaseDate: z.string().transform((s) => new Date(s)),
  purchaseValue: z.number().positive(),
  warrantyExpiry: z.string().transform((s) => new Date(s)),
  amcValidUpto: z.string().transform((s) => new Date(s)),
  photos: z.array(z.string()).optional()
});

Async function generateAssetCode(propertyId) {
  Const property = await prisma.property.findUnique({
    Where: { id: propertyId },
    Select: { propertyPrefix: true }
  });
  If (!property) {
    Throw new Error(‘Property not found’);
  }

  Const prefix = property.propertyPrefix;

  Const assets = await prisma.asset.findMany({
    Where: { propertyId },
    Select: { assetCode: true },
    orderBy: { createdAt: ‘desc’ }
  });

  Let nextSeq = 1;
  If (assets.length > 0) {
    Const lastCode = assets[0].assetCode;
    Const parts = lastCode.split(‘-‘);
    Const lastSeqStr = parts[parts.length – 1];
    Const lastSeq = parseInt(lastSeqStr, 10);
    nextSeq = lastSeq + 1;
  }

  Const seqPadded = String(nextSeq).padStart(4, ‘0’);
  Return `${prefix}-AST-${seqPadded}`;
}

Router.get(‘/’, async (req, res) => {
  Const { propertyId } = req.query;
  If (!propertyId) {
    Return res.status(400).json({ error: ‘propertyId query param required’ });
  }

  Const assets = await prisma.asset.findMany({
    Where: { propertyId },
    Include: {
      Property: { select: { id: true, name: true } },
      Creator: { select: { id: true, name: true } }
    },
    orderBy: { createdAt: ‘desc’ }
  });

  Res.json(assets);
});

Router.get(‘/:id’, async (req, res) => {
  Const asset = await prisma.asset.findUnique({
    Where: { id: req.params.id },
    Include: {
      Property: true,
      Creator: true,
      Renewals: true,
      Repairs: true
    }
  });
  If (!asset) return res.status(404).json({ error: ‘Asset not found’ });
  Res.json(asset);
});

Router.post(‘/’, async (req, res) => {
  Try {
    Const body = createAssetSchema.parse(req.body);

    Const assetCode = await generateAssetCode(body.propertyId);

    Const asset = await prisma.asset.create({
      Data: {
        …body,
        assetCode,
        status: ‘ACTIVE’,
        createdBy: req.user.id
      },
      Include: { property: true }
    });

    Await logAudit(
      Req.user.id,
      ‘CREATE’,
      ‘Asset’,
      Asset.id,
      { action: ‘created’, assetCode, data: body }
    );

    Res.status(201).json(asset);
  } catch € {
    Res.status(400).json({ error: ‘Invalid input’, details: e.message });
  }
});

Router.put(‘/:id’, async (req, res) => {
  Const asset = await prisma.asset.findUnique({ where: { id: req.params.id } });
  If (!asset) return res.status(404).json({ error: ‘Asset not found’ });

  Const body = createAssetSchema.partial().parse(req.body);

  Const updated = await prisma.asset.update({
    Where: { id: asset.id },
    Data: body
  });

  Await logAudit(
    Req.user.id,
    ‘UPDATE’,
    ‘Asset’,
    Asset.id,
    { action: ‘updated’, changes: body }
  );

  Res.json(updated);
});

Router.post(‘/:id/renewals’, async (req, res) => {
  Const asset = await prisma.asset.findUnique({ where: { id: req.params.id } });
  If (!asset) return res.status(404).json({ error: ‘Asset not found’ });

  Const schema = z.object({
    renewalType: z.string(),
    startDate: z.string().transform((s) => new Date(s)),
    endDate: z.string().transform((s) => new Date(s)),
    cost: z.number().positive(),
    vendor: z.string().optional(),
    billUrl: z.string().url().optional(),
    notes: z.string().optional()
  });

  Const data = schema.parse(req.body);

  Const renewal = await prisma.assetRenewal.create({
    Data: {
      assetId: asset.id,
      …data
    }
  });

  Await logAudit(
    Req.user.id,
    ‘CREATE’,
    ‘AssetRenewal’,
    Renewal.id,
    { action: ‘created’, assetId: asset.id, data }
  );

  Res.status(201).json(renewal);
});

Router.post(‘/:id/repairs’, async (req, res) => {
  Const asset = await prisma.asset.findUnique({ where: { id: req.params.id } });
  If (!asset) return res.status(404).json({ error: ‘Asset not found’ });

  Const schema = z.object({
    repairDate: z.string().transform((s) => new Date(s)),
    description: z.string(),
    billUrl: z.string().url().optional(),
    cost: z.number().positive(),
    vendor: z.string().optional(),
    performedBy: z.string().optional()
  });

  Const data = schema.parse(req.body);

  Const repair = await prisma.assetRepair.create({
    Data: {
      assetId: asset.id,
      …data
    }
  });

  Await logAudit(
    Req.user.id,
    ‘CREATE’,
    ‘AssetRepair’,
    Repair.id,
    { action: ‘created’, assetId: asset.id, data }
  );

  Res.status(201).json(repair);
});

Export default router;
