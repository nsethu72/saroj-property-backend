Import express from ‘express’;
Import { prisma } from ‘../index.js’;
Import { z } from ‘zod’;
Import { logAudit } from ‘../utils/helpers.js’;

Const router = express.Router();

Const generateScheduleSchema = z.object({
  leaseId: z.string(),
  startDate: z.string().transform((s) => new Date(s)),
  months: z.number().int().positive(),
  monthlyAmount: z.number().positive()
});

Router.post(‘/generate-schedule’, async (req, res) => {
  Try {
    Const { leaseId, startDate, months, monthlyAmount } = generateScheduleSchema.parse(req.body);

    Const lease = await prisma.lease.findUnique({ where: { id: leaseId } });
    If (!lease) return res.status(404).json({ error: ‘Lease not found’ });

    Const schedules = [];
    For (let I = 0; I < months; i++) {
      Const due = new Date(startDate);
      Due.setMonth(due.getMonth() + i);
      Schedules.push({
        leaseId,
        dueDate: due,
        amount: monthlyAmount,
        status: ‘UNPAID’
      });
    }

    Const created = await prisma.rentSchedule.createMany({
      Data: schedules
    });

    Await logAudit(
      Req.user.id,
      ‘CREATE’,
      ‘RentSchedule’,
      leaseId,
      { action: ‘generated_schedule’, months, monthlyAmount }
    );

    Res.json({ count: created.count, schedules });
  } catch € {
    Res.status(400).json({ error: ‘Invalid input’, details: e.message });
  }
});

Router.get(‘/schedule’, async (req, res) => {
  Const { leaseId } = req.query;
  If (!leaseId) {
    Return res.status(400).json({ error: ‘leaseId query param required’ });
  }

  Const schedules = await prisma.rentSchedule.findMany({
    Where: { leaseId },
    Include: { lease: { select: { id: true, flatNumber: true } } },
    orderBy: { dueDate: ‘asc’ }
  });

  Res.json(schedules);
});

Router.post(‘/payments’, async (req, res) => {
  Const schema = z.object({
    rentScheduleId: z.string(),
    paidAmount: z.number().positive(),
    paidAt: z.string().transform((s) => new Date(s)).optional()
  });

  Try {
    Const { rentScheduleId, paidAmount, paidAt } = schema.parse(req.body);

    Const schedule = await prisma.rentSchedule.findUnique({
      Where: { id: rentScheduleId },
      Include: { lease: true }
    });
    If (!schedule) return res.status(404).json({ error: ‘Rent schedule not found’ });

    Const newPaidAmount = (schedule.paidAmount || 0) + paidAmount;
    Let newStatus = schedule.status;

    If (newPaidAmount <= 0) {
      newStatus = ‘UNPAID’;
    } else if (newPaidAmount >= schedule.amount) {
      newStatus = ‘PAID’;
    } else {
      newStatus = ‘PARTIAL’;
    }

    Const updated = await prisma.rentSchedule.update({
      Where: { id: rentScheduleId },
      Data: {
        paidAmount: newPaidAmount,
        paidAt: paidAt || new Date(),
        status: newStatus
      },
      Include: { lease: true }
    });

    Await logAudit(
      Req.user.id,
      ‘UPDATE’,
      ‘RentSchedule’,
      rentScheduleId,
      { action: ‘payment_recorded’, paidAmount, newStatus }
    );

    Res.json(updated);
  } catch € {
    Res.status(400).json({ error: ‘Invalid input’, details: e.message });
  }
});

Router.get(‘/reminders/config’, async (req, res) => {
  Const { propertyId } = req.query;
  Const config = await prisma.reminderConfig.findFirst({
    Where: { propertyId: propertyId || null }
  });
  Res.json(config || { propertyId: propertyId || null, daysBeforeDue: 3, daysAfterDue: 1, enabled: true });
});

Router.put(‘/reminders/config’, async (req, res) => {
  Const schema = z.object({
    propertyId: z.string().optional(),
    daysBeforeDue: z.number().int().nonnegative().optional(),
    daysAfterDue: z.number().int().nonnegative().optional(),
    enabled: z.boolean().optional()
  });

  Const data = schema.parse(req.body);
  Const propertyId = data.propertyId || null;

  Const existing = await prisma.reminderConfig.findFirst({
    Where: { propertyId }
  });

  Let config;
  If (existing) {
    Config = await prisma.reminderConfig.update({
      Where: { id: existing.id },
      Data
    });
  } else {
    Config = await prisma.reminderConfig.create({
      Data: {
        propertyId,
        daysBeforeDue: data.daysBeforeDue ?? 3,
        daysAfterDue: data.daysAfterDue ?? 1,
        enabled: data.enabled ?? true
      }
    });
  }

  Await logAudit(
    Req.user.id,
    ‘UPDATE’,
    ‘ReminderConfig’,
    Config.id,
    { action: ‘updated_config’, data }
  );

  Res.json(config);
});

Export default router;
