Import express from ‘express’;
Import { prisma } from ‘../index.js’;
Import { z } from ‘zod’;
Import { logAudit } from ‘../utils/helpers.js’;

Const router = express.Router();

Const createFixtureSchema = z.object({
  unitId: z.string(),
  name: z.string(),
  category: z.string(),
  quantity: z.number().int().positive(),
  condition: z.string(),
  notes: z.string().optional(),
  photos: z.array(z.string()).optional()
});

Router.get(‘/’, async (req, res) => {
  Const { unitId } = req.query;
  If (!unitId) {
    Return res.status(400).json({ error: ‘unitId query param required’ });
  }

  Const fixtures = await prisma.fixture.findMany({
    Where: { unitId },
    Include: { unit: true, creator: true },
    orderBy: { createdAt: ‘desc’ }
  });

  Res.json(fixtures);
});

Router.post(‘/’, async (req, res) => {
  Try {
    Const body = createFixtureSchema.parse(req.body);
    Const fixture = await prisma.fixture.create({
      Data: {
        …body,
        createdBy: req.user.id
      },
      Include: { unit: true }
    });

    Await logAudit(
      Req.user.id,
      ‘CREATE’,
      ‘Fixture’,
      Fixture.id,
      { action: ‘created’, data: body }
    );

    Res.status(201).json(fixture);
  } catch € {
    Res.status(400).json({ error: ‘Invalid input’, details: e.message });
  }
});

Export default router;
